25typedef enum

76typedef struct
77{
78 arq_event_id_t id;

80 /* RX events: the session id, the carrier the frame came on, and the
81 * frame itself (RX_CAROUSEL). */
82 uint8_t session_id;
83 int mode;
84 uint8_t payload[1280]; /* >= largest user payload: QAM16C2
85 * carries 1213 - 8 = 1205 bytes */
86 size_t payload_len;
87
88 /* RX_CAROUSEL: the control decoder (DATAC16) produced it; otherwise the
89 * payload decoder, in `mode`. RX_ACCEPT: the rung the callee starts our
90 * rounds on (0..6, or ARQ_ACCEPT_LEVEL_CTL_DEAF), its session nonce, and
91 * the CRC16 of the callsign we called, which the FSM checks. */
92 bool from_control;
93 int car_level;
94 uint16_t car_nonce;
95 uint16_t car_called_crc;
96
97 /* Local receive SNR at the time the frame was decoded (dB, 0 = unknown):
98 * a CALL's sets the start rung the ACCEPT names. */
99 float rx_snr;
100 /* RX_CAROUSEL: how long ago the frame ended when it was decoded. */
101 uint32_t rx_age_ms;
102
103 /* RX_PATTERN: the BREAK kind, else the session ACK. */
104 bool pattern_break;
105
106 /* APP_DISCONNECT from the host's ABORT: drop what is still in flight and
107 * go at once, rather than drain it first as DISCONNECT does (#218). */
108 bool abort;
109
110 /* Call setup */
111 char remote_call[CALLSIGN_MAX_SIZE];
112 /* For an incoming CALL: which of OUR callsigns (primary or a secondary)
113 * the frame's DST CRC16 matched — i.e. the SSID the caller dialed. Empty
114 * when not applicable; the callee reports this as the connection's local
115 * address so a station listening on multiple SSIDs shows the dialed one. */
116 char local_call[CALLSIGN_MAX_SIZE];
117} arq_event_t;

119/* ======================================================================
120 * Session structure — replaces arq_ctx_t
121 *
122 * All state is in this struct; no hidden global flags.
123 * Monotonic timestamps are uint64_t milliseconds from hermes_log startup.
124 * ====================================================================== */
125
126typedef struct
127{
128 /* --- State machine fields --- */
129 arq_conn_state_t conn_state; /* connection state */
130
131 /* --- Identifiers --- */
132 uint8_t session_id; /* random byte chosen by caller */
133 int call_sends_done; /* CALLs sent this attempt (escalation) */
134 int call_carrier; /* the carrier the last CALL went out on */
135 int call_rx_mode; /* the carrier the CALL we answer came on */
136 char remote_call[CALLSIGN_MAX_SIZE];
137 char local_call[CALLSIGN_MAX_SIZE]; /* our dialed callsign/SSID for an
138 * accepted incoming CALL (empty =>
139 * fall back to the primary) */
140
141 /* --- Mode / speed --- */
142 int payload_mode; /* MY data TX mode (ISS); per-direction,
143 * independent of peer's TX mode */
144 int control_mode; /* always FREEDV_MODE_DATAC16 */
145 int initial_payload_mode; /* startup payload mode (= broadcast RX
146 * mode); restored on disconnect so the
147 * payload decoder matches broadcast */
148 int peer_tx_mode; /* my payload RX decoder's mode: the
149 * carousel binds it to the peer's
150 * rung, idle restores the listen mode */
151
152 /* --- Retry/timeout bookkeeping --- */
153 int tx_retries_left; /* retries remaining for current frame.
154 * In ACCEPTING this counts TIMER_RETRY
155 * slots, not transmissions: a slot may be
156 * spent silently when the RX window closes
157 * without a CALL (see fsm_accepting). */
158 uint64_t state_enter_ms; /* when current conn_state was entered */
159
160 /* When a decoder was last holding sync on an incoming burst, i.e. when the
161 * peer was last observed transmitting. Written by the modem RX thread via
162 * arq_update_link_metrics() under g_sess_lock, read by the FSM; used to
163 * avoid keying into the peer's frame. */
164 uint64_t last_rx_sync_ms;
165 /* A decoder caught a frame's preamble: the frame is on the air until
166 * this, whatever its sync does meanwhile (arq_note_rx_preamble). */
167 uint64_t rx_frame_busy_until_ms;
168 uint64_t nav_busy_until_ms; /* a NAV header heard: the peer's keydown runs until then */
169 uint8_t disc_defer_count; /* consecutive busy-channel deferrals
170 * of our DISCONNECT */
171 bool tx_active; /* we are keyed (TX_STARTED seen, no
172 * TX_COMPLETE yet) */
173 uint64_t last_tx_end_ms; /* when our last transmission ended */
174 bool host_released; /* the application ended this session
175 * (DISCONNECT/ABORT): nothing more is
176 * delivered to it until a new one.
177 * Only APP_DISCONNECT (DISCONNECT,
178 * ABORT, control client gone) sets
179 * it: it is the one teardown that stays
180 * CONNECTED (draining the backlog)
181 * after the host is told DISCONNECTED.
182 * Peer DISCONNECT, timeouts and LISTEN
183 * OFF leave CONNECTED at once, so
184 * nothing is delivered after them. */
185 int local_snr_x10; /* local SNR EMA * 10 */
186
187 /* --- Data bookkeeping --- */
188 int tx_inflight_bytes; /* bytes the carousel holds unconfirmed */
189 int pending_burst_frames; /* frames accumulated for current PTT burst;
190 * written by cb_send_tx_frame under g_sess_lock
191 * (replaces the former function-static so that
192 * the cmd-bridge SEND_CQ path and the FSM event
193 * loop do not race on a shared static) */
194
195 /* --- Connection lifecycle --- */
196 bool listen_enabled; /* app listen intent (LISTEN ON/OFF):
197 * selects the post-call idle status,
198 * LISTENING vs DISCONNECTED, so an ARQ
199 * call always returns to where it was */
200
201 /* --- Connect handshake --- */
202 bool accept_fallback; /* LISTENING was entered because our ACCEPT
203 * retries ran out: the caller may still
204 * have heard it, so its first carousel
205 * frame may complete this session. Only
206 * then. */
207 bool accept_tx_pending; /* the pending TIMER_RETRY is an ACCEPT
208 * answering a CALL we actually heard,
209 * not the RX-window timer. Only the
210 * former may key the transmitter -- see
211 * fsm_accepting's TIMER_RETRY. */
212
213 /* --- Teardown flags --- */
214 bool deferred_listen_off; /* LISTEN OFF received during grace period;
215 * will be honoured once the grace expires */
216 bool pending_disconnect_notify;/* defer notify_disconnected until TX done */
217 bool disconnect_sent; /* our DISCONNECT is on the air this teardown */
218 bool pending_disconnect_reply; /* our reply to the peer's DISCONNECT is
219 * due when the reply guard expires */
220 bool pending_disconnect; /* APP_DISCONNECT deferred until TX buf empty */
221 bool pending_connect; /* CONNECT arrived while DISCONNECTING: place
222 * the call once the teardown completes */
223 char pending_connect_call[CALLSIGN_MAX_SIZE];
224 uint64_t disconnect_deadline_ms; /* absolute time by which a deferred
225 * APP_DISCONNECT must resolve into a
226 * clean teardown. 0 = none armed. */
227
228 uint64_t last_rx_ms; /* last frame decoded from the peer */
229
230 /* --- Carousel data plane (carousel.h) --- *
231 * car_active while CONNECTED (and through a callee's ending); deadline_ms
232 * then mirrors the carousel's next deadline (ARQ_EV_TIMER_CAROUSEL). crc_seed rides every carousel frame's CRC16. */
233 car_t *car; /* allocated by arq_fsm_init */
234 bool car_active;
235 bool car_session; /* this session ran the carousel (until idle) */
236 bool car_ending; /* the callee ends it: DISCONNECT in its next slot */
237 uint64_t car_end_deadline_ms; /* ...or gives up then, silently */
238 uint16_t crc_seed;
239 uint16_t car_nonce; /* the callee's, in the ACCEPT: part of crc_seed */
240 int car_rx_level; /* start rung for the peer's direction:
241 * the SNR measured here */
242 int car_tx_level; /* ...for ours, from the peer (-1: none) */
243 bool car_ctl_deaf; /* connect: I heard the peer below the control mode */
244 bool car_peer_ctl_deaf; /* connect: ...and it, me (the CALL came on MFSK) */
245 bool ctl_floor_at_stop; /* the carousel's peer_ctl_deaf when it stopped,
246 * for the DISCONNECT exchange after it */
247 uint64_t car_last_rx_ms; /* last carousel frame from the peer */
248
249 /* --- Timer mechanism --- */
250 uint64_t deadline_ms; /* absolute monotonic deadline */
251 arq_event_id_t deadline_event; /* event to fire when deadline fires */
252} arq_session_t;
