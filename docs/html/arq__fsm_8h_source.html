25typedef enum
26{
27 ARQ_CONN_DISCONNECTED = 0, /* no session; idle */
28 ARQ_CONN_LISTENING = 1, /* waiting for incoming CALL frame */
29 ARQ_CONN_CALLING = 2, /* outgoing CALL sent; awaiting ACCEPT */
30 ARQ_CONN_ACCEPTING = 3, /* ACCEPT sent; awaiting the caller's first frame */
31 ARQ_CONN_CONNECTED = 4, /* the carousel runs the session */
32 ARQ_CONN_DISCONNECTING = 5, /* DISCONNECT frame being exchanged */
33 ARQ_CONN__COUNT
34} arq_conn_state_t;

36/* The step a teardown has reached. Each value belongs to one connection
37 * state, and every state change resets it (sess_enter): nothing of one
38 * teardown is left for the next session to trip on. */
39typedef enum
40{
41 ARQ_TD_NONE = 0,
42 ARQ_TD_LISTEN_OFF, /* ACCEPTING: a LISTEN OFF inside the grace period,
43 * honoured when it expires or once connected */
44 ARQ_TD_DRAINING, /* CONNECTED: the application's DISCONNECT waits
45 * for delivery, until teardown_deadline_ms */
46 ARQ_TD_SENT, /* DISCONNECTING: our DISCONNECT is on the air */
47 ARQ_TD_REPLYING, /* DISCONNECTED: the reply to the peer's
48 * DISCONNECT is due when the reply guard expires */
49 ARQ_TD_NOTIFYING, /* DISCONNECTED: the reply is out; the host hears
50 * DISCONNECTED when it has finished (TX_COMPLETE),
51 * so data_rx_buffer_arq drains to it first */
52} arq_teardown_t;

94typedef struct
95{
96 arq_event_id_t id;
97
98 /* RX events: the session id, the carrier the frame came on, and the
99 * frame itself (RX_CAROUSEL). */
100 uint8_t session_id;
101 int mode;
102 uint8_t payload[1280]; /* >= largest user payload: QAM16C2
103 * carries 1213 - 8 = 1205 bytes */
104 size_t payload_len;
105
106 /* RX_CAROUSEL: the control decoder (DATAC16) produced it; otherwise the
107 * payload decoder, in `mode`. RX_ACCEPT: the rung the callee starts our
108 * rounds on (0..6), whether it hears us below the control mode, the SNR
109 * it read our CALL at (0: unknown), its session nonce, and the CRC16 of
110 * the callsign we called, which the FSM checks. */
111 bool from_control;
112 int bw_hz; /* RX_CALL / RX_ACCEPT: the bandwidth it advertises */
113 int car_level;
114 bool car_ctl_deaf;
115 float car_call_snr;
116 uint16_t car_nonce;
117 uint16_t car_called_crc;
118
119 /* Local receive SNR at the time the frame was decoded (dB, 0 = unknown):
120 * a CALL's sets the start rung the ACCEPT names. */
121 float rx_snr;
122 /* RX_CAROUSEL: how long ago the frame ended when it was decoded. */
123 uint32_t rx_age_ms;
124
125 /* RX_PATTERN: the BREAK kind, else the session ACK. */
126 bool pattern_break;
127
128 /* APP_DISCONNECT from the host's ABORT: drop what is still in flight and
129 * go at once, rather than drain it first as DISCONNECT does (#218). */
130 bool abort;
131
132 /* Call setup */
133 char remote_call[CALLSIGN_MAX_SIZE];
134 /* For an incoming CALL: which of OUR callsigns (primary or a secondary)
135 * the frame's DST CRC16 matched — i.e. the SSID the caller dialed. Empty
136 * when not applicable; the callee reports this as the connection's local
137 * address so a station listening on multiple SSIDs shows the dialed one. */
138 char local_call[CALLSIGN_MAX_SIZE];
139} arq_event_t;

141/* ======================================================================
142 * Session structure — replaces arq_ctx_t
143 *
144 * All state is in this struct; no hidden global flags.
145 * Monotonic timestamps are uint64_t milliseconds from hermes_log startup.
146 * ====================================================================== */
147
148typedef struct
149{
150 /* --- State machine fields --- */
151 arq_conn_state_t conn_state; /* connection state */
152
153 /* --- Identifiers --- */
154 uint8_t session_id; /* random byte chosen by caller */
155 int call_sends_done; /* CALLs sent this attempt (escalation) */
156 int call_carrier; /* the carrier the last CALL went out on */
157 int call_rx_mode; /* the carrier the CALL we answer came on */
158 char remote_call[CALLSIGN_MAX_SIZE];
159 char local_call[CALLSIGN_MAX_SIZE]; /* our dialed callsign/SSID for an
160 * accepted incoming CALL (empty =>
161 * fall back to the primary) */
162
163 /* --- Mode / speed --- */
164 int payload_mode; /* MY data TX mode (ISS); per-direction,
165 * independent of peer's TX mode */
166 int control_mode; /* always FREEDV_MODE_DATAC16 */
167 int initial_payload_mode; /* startup payload mode (= broadcast RX
168 * mode); restored on disconnect so the
169 * payload decoder matches broadcast */
170 int peer_tx_mode; /* my payload RX decoder's mode: the
171 * carousel binds it to the peer's
172 * rung, idle restores the listen mode */
173
174 /* --- Retry/timeout bookkeeping --- */
175 int tx_retries_left; /* retries remaining for current frame.
176 * In ACCEPTING this counts TIMER_RETRY
177 * slots, not transmissions: a slot may be
178 * spent silently when the RX window closes
179 * without a CALL (see fsm_accepting). */
180 uint64_t state_enter_ms; /* when current conn_state was entered */
181
182 /* When a decoder was last holding sync on an incoming burst, i.e. when the
183 * peer was last observed transmitting. Written by the modem RX thread via
184 * arq_update_link_metrics() under g_sess_lock, read by the FSM; used to
185 * avoid keying into the peer's frame. */
186 uint64_t last_rx_sync_ms;
187 /* A decoder caught a frame's preamble: the frame is on the air until
188 * this, whatever its sync does meanwhile (arq_note_rx_preamble). */
189 uint64_t rx_frame_busy_until_ms;
190 uint64_t nav_busy_until_ms; /* a NAV header heard: the peer's keydown runs until then */
191 uint8_t defer_count; /* consecutive busy-channel deferrals
192 * of our DISCONNECT, or our CALL */
193 bool tx_active; /* we are keyed (TX_STARTED seen, no
194 * TX_COMPLETE yet) */
195 uint64_t last_tx_end_ms; /* when our last transmission ended */
196 bool host_released; /* the application ended this session
197 * (DISCONNECT/ABORT): nothing more is
198 * delivered to it until a new one.
199 * Only APP_DISCONNECT (DISCONNECT,
200 * ABORT, control client gone) sets
201 * it: it is the one teardown that stays
202 * CONNECTED (draining the backlog)
203 * after the host is told DISCONNECTED.
204 * Peer DISCONNECT, timeouts and LISTEN
205 * OFF leave CONNECTED at once, so
206 * nothing is delivered after them. */
207 int local_snr_x10; /* local SNR EMA * 10 */
208
209 /* --- Data bookkeeping --- */
210 int tx_inflight_bytes; /* bytes the carousel holds unconfirmed */
211 int pending_burst_frames; /* frames accumulated for current PTT burst;
212 * written by cb_send_tx_frame under g_sess_lock
213 * (replaces the former function-static so that
214 * the cmd-bridge SEND_CQ path and the FSM event
215 * loop do not race on a shared static) */
216
217 /* --- Connection lifecycle --- */
218 bool listen_enabled; /* app listen intent (LISTEN ON/OFF):
219 * selects the post-call idle status,
220 * LISTENING vs DISCONNECTED, so an ARQ
221 * call always returns to where it was */
222
223 /* --- Connect handshake --- */
224 bool accept_fallback; /* LISTENING was entered because our ACCEPT
225 * retries ran out: the caller may still
226 * have heard it, so its first carousel
227 * frame may complete this session. Only
228 * then. */
229 bool accept_tx_pending; /* the pending TIMER_RETRY is an ACCEPT
230 * answering a CALL we actually heard,
231 * not the RX-window timer. Only the
232 * former may key the transmitter -- see
233 * fsm_accepting's TIMER_RETRY. */
234
235 /* --- Teardown --- */
236 arq_teardown_t teardown; /* the step reached (arq_teardown_t) */
237 uint64_t teardown_deadline_ms; /* when that step gives up: a held
238 * DISCONNECT tears down regardless
239 * (CONNECTED), the callee ending stops
240 * waiting for an answer (DISCONNECTING) */
241 char pending_connect_call[CALLSIGN_MAX_SIZE];
242 /* a CONNECT that came during the
243 * teardown, placed once it completes;
244 * empty = none */
245
246 uint64_t last_rx_ms; /* last frame decoded from the peer */
247
248 /* --- Carousel data plane (carousel.h) --- *
249 * car_active while CONNECTED (and through a callee's ending); deadline_ms
250 * then mirrors the carousel's next deadline (ARQ_EV_TIMER_CAROUSEL). crc_seed rides every carousel frame's CRC16. */
251 car_t *car; /* allocated by arq_fsm_init */
252 bool car_active;
253 bool car_session; /* this session ran the carousel (until idle) */
254 uint16_t crc_seed;
255 uint16_t car_nonce; /* the callee's, in the ACCEPT: part of crc_seed */
256 int car_rx_level; /* start rung for the peer's direction:
257 * the SNR measured here */
258 int car_tx_level; /* ...for ours, from the peer (-1: none) */
259 bool car_ctl_deaf; /* connect: I heard the peer below the control mode */
260 bool car_peer_ctl_deaf; /* connect: ...and it, me (the CALL came on MFSK) */
261 float car_call_snr; /* connect: the SNR the callee read the CALL at
262 * (the callee: its own; the caller: from the
263 * ACCEPT), 0 unknown */
264 float car_my_snr; /* connect: the SNR I read the peer's connect
265 * frame at (the CALL, the ACCEPT), 0 unknown */
266 bool ctl_floor_at_stop; /* the carousel's peer_ctl_deaf when it stopped,
267 * for the DISCONNECT exchange after it */
268 int floor_mode; /* the session's MFSK floor (arq_floor_mode
269 * of its bandwidth), from its seed on */
270 uint64_t car_last_rx_ms; /* last carousel frame from the peer */
271
272 /* --- Timer mechanism --- */
273 uint64_t deadline_ms; /* absolute monotonic deadline */
274 arq_event_id_t deadline_event; /* event to fire when deadline fires */
275} arq_session_t;
