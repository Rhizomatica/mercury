86/* A keydown of separate bursts -- each its own preamble, frame and postamble,
87 * in its own mode -- with a silence before each but the first (the carousel
88 * data plane's rounds). Heap-allocated by the enqueuer; whoever dequeues the
89 * action frees it. */
90/* A NAV header (mfsk.h) and the silence after it, ahead of a keydown's first
91 * frame: what the carousel adds to a headed keydown's lead. */
92#define ARQ_NAV_GAP_MS 60
93#define ARQ_NAV_LEAD_MS (640 + ARQ_NAV_GAP_MS)
94#define ARQ_KEYDOWN_FRAMES 17
95#define ARQ_KEYDOWN_FRAME_MAX 1280
96typedef struct
97{
98 int n;
99 uint16_t crc_seed; /* XORed into every frame's CRC16 (0: plain) */
100 bool nav; /* open with a NAV header (ARQ_NAV_LEAD_MS) */
101 struct
102 {
103 int mode;
104 uint32_t gap_ms;
105 size_t len;
106 uint8_t bytes[ARQ_KEYDOWN_FRAME_MAX];
107 } f[ARQ_KEYDOWN_FRAMES];
108} arq_keydown_t;
