17/* ======================================================================
18 * Plain session frame header (8 bytes): the DISCONNECT's. A session's other
19 * frames are the carousel's (carousel.h), CALL/ACCEPT have their own layout
20 * (below).
21 *
22 * Byte 0: framer byte — set/read by write_frame_header()/parse_frame_header()
23 * bits [7:5] = packet_type (PACKET_TYPE_ARQ_CONTROL)
24 * bits [4:0] = extension field (0)
25 * Byte 1: subtype — arq_subtype_t
26 * Byte 2: flags — 0
27 * Byte 3: session_id — the session's id, chosen by the caller
28 * Byte 4: tx_seq — 0
29 * Byte 5: rx_ack_seq — 0
30 * Byte 6: snr_raw — local RX SNR; 0=unknown, else round(dB) + 128
31 * Byte 7: ack_delay — 0
32 * ====================================================================== */
33

43/* CONNECT frames (CALL/ACCEPT) compact layout — 14 bytes, DATAC16 only.
44 * Uses PACKET_TYPE_ARQ_CALL2 in the framer byte.
45 *
46 * Byte 0: framer byte (PACKET_TYPE_ARQ_CALL2 | BW token, minor revision,
47 * TX-MFSK-only bit, set by write_frame_header)
48 * Byte 1: connect_meta = (session_id & 0x7F) | (is_accept ? 0x80 : 0x00)
49 * Bytes 2-3: CRC16-CCITT of DST callsign (little-endian) — for local validation
50 * Bytes 4-13: arithmetic_encode(SRC callsign only) — 10 bytes, fits callsigns up to ~14 chars
51 */
52#define ARQ_CONNECT_SESSION_IDX 1
53#define ARQ_CONNECT_PAYLOAD_IDX 2
54#define ARQ_CONNECT_SESSION_MASK 0x7F
55#define ARQ_CONNECT_ACCEPT_FLAG 0x80
56#define ARQ_CONTROL_FRAME_SIZE 14
57#define ARQ_CONNECT_META_SIZE 2 /* framer byte + connect_meta byte */
58#define ARQ_CONNECT_MAX_ENCODED (ARQ_CONTROL_FRAME_SIZE - ARQ_CONNECT_META_SIZE)
59#define ARQ_CONNECT_DST_CRC_SIZE 2 /* CRC16-CCITT of DST at bytes [2..3], little-endian */
60#define ARQ_CONNECT_SRC_MAX_ENCODED (ARQ_CONNECT_MAX_ENCODED - ARQ_CONNECT_DST_CRC_SIZE) /* 10 */

126/* Timing constants shared across modules. The two guards are measured, not
127 * tuned, and are no longer configuration: they bundle the radio's TX->RX
128 * switch with the audio backend's buffering, which nobody can tune blind. */
129#define ARQ_CHANNEL_GUARD_MS 700 /* IRS response guard after frame decode.
130 * OFDM decode fires ~200ms before sender
131 * PTT-OFF, so effective gap at sender is
132 * (guard - 200ms) ~= 500ms. Radio needs
133 * ~340ms for TX->RX switch -> 160ms margin
134 * for preamble detection. At 500ms the
135 * effective gap was ~300ms, causing ~50%
136 * ACK loss on DATAC1 (< 340ms switch). */
137
138#define ARQ_ISS_POST_ACK_GUARD_MS 900 /* ISS guard before resuming DATA TX
139 * after receiving an ACK from the IRS.
140 * Larger than ARQ_CHANNEL_GUARD_MS:
141 * ack_rx fires ~168ms before IRS PTT-OFF,
142 * so the effective gap at IRS is only
143 * (guard + 100ms head) - 168ms.
144 * At 500ms: gap=432ms, too tight for
145 * DATAC1 re-sync after IRS ACK TX.
146 * At 900ms: gap=832ms -- 492ms of clear
147 * air before the DATAC1 preamble. */
148#define ARQ_ACCEPT_RX_WINDOW_MS 9000 /* ACCEPTING RX window after ACCEPT TX:
149 * ISS_guard(900)+DATAC15(4400)+margin(3700)
150 * Old value 7000 left only ~300ms margin
151 * and raced with TIMER_RETRY, causing
152 * 3-4 wasted ACCEPT retries (~28s). */
153/* Grace period after entering CALLING/ACCEPTING (i.e. after PENDING is sent)
154 * during which LISTEN OFF is deferred. Scanning hosts (BPQ32) need a moment to
155 * process PENDING and cancel their own dwell timers; without this grace, a
156 * LISTEN OFF that crosses PENDING on the wire cancels the handshake — the host
157 * moves to the next channel and the reply goes out on the wrong frequency.
158 * 2 s is conservative: ~0.5 s of TCP+host processing with headroom. */
159#define ARQ_LISTEN_OFF_GRACE_MS 2000
160#define ARQ_CALL_RETRY_SLOTS_DEFAULT 4 /* CALL retries before giving up */
161/* CALLs on the fast control mode before the caller escalates to the MFSK
162 * floor; after that it alternates MFSK and the control mode (#235: under
163 * fading an MFSK CALL is no surer than a DATAC16 one at -9..-11 dB, and all-
164 * MFSK lost connects DATAC16 made; MFSK is what connects at -13 dB). A good
165 * link connects exactly as quickly as before. */
166#define ARQ_CALL_FAST_SLOTS_DEFAULT 2
167/* ...and at most this many, whatever RETRIES says: each fast CALL puts the
168 * first MFSK CALL ~12 s later, and a path only the floor carries connects
169 * that much later (specs/connect: 112 s at 2, ~134 s at 4, 205 s at 10, where
170 * BPQ32 gave up after ~87 s, #442). */
171#define ARQ_CALL_FAST_SLOTS_MAX 4
172/* Runtime value: RETRIES N above the default raises it to N/2, capped at
173 * ARQ_CALL_FAST_SLOTS_MAX, so a long retry count buys more fast CALLs (what a
174 * scanning BBS hears while it dwells on the channel; at any bandwidth), not
175 * only more slow MFSK steps (#442). */

215/* How long a session hears nothing before its control moves to the MFSK floor

247/* The silence between two frames of one keydown (the carousel's poll and its
248 * first burst). The peer is already receiving, so this is not a turnaround:
249 * it only has to let the peer's control decoder report the first frame and
250 * its payload decoder drop any sync taken off it before the next preamble
251 * arrives. Measured, not tuned: see the one-keydown bench. */
252#define ARQ_ACK_DATA_GAP_MS 300

254/* Listen before talk. "Busy" is a decoder holding sync on an incoming burst
255 * (arq_update_link_metrics()); HOLD_MS is how stale a sync observation may be
256 * and still count as busy. The modem updates it once per RX chunk, so this
257 * only covers scheduling jitter, not a whole frame: sync keeps being
258 * re-observed for as long as the burst lasts.
259 *
260 * A deferred transmission (the DISCONNECT) waits in DEFER_MS steps, at most
261 * DEFER_MAX of them: false sync is real (a payload decoder can latch onto a
262 * control burst), and a decoder stuck in sync must not block a transmission
263 * forever, so after that we key anyway and take the collision. */
264#define ARQ_CHANNEL_SYNC_HOLD_MS 250
265#define ARQ_CHANNEL_DEFER_MS 500
266#define ARQ_CHANNEL_DEFER_MAX 20 /* ~10 s, then key regardless */
267/* A CALL held for a frame (arq_fsm.c, call_must_wait) waits this long past
268 * the frame's end, so that the frame can decode first: an ACCEPT does, and a
269 * CALL keyed just before it went out from a caller already connected. The
270 * floor's decode allowance (carousel.c D_MFSK_MS 4500; ~3.7 s on air) and one
271 * deferral step: a decode at the allowance's very end otherwise ties with the
272 * CALL (specs/connect/ConnectCarrier.tla, DW). */
273#define ARQ_CALL_DECODE_WAIT_MS (4500 + ARQ_CHANNEL_DEFER_MS)

406/* --- CALL/ACCEPT of Mercury 2.0 (PACKET_TYPE_ARQ_CALL2) ---
407 *
408 * 14 bytes (a DATAC16 frame), padded to the frame on MFSK. Byte 0 is the
409 * framer: type CALL2, and in its extension the bandwidth token (2 bits), the
410 * 2.x minor revision (2 bits, 1) and the TX-MFSK-only bit. Every 1.9.x
411 * release drops the type,
412 * and a CALL of theirs (PACKET_TYPE_ARQ_CALL) addressed to this station is
413 * logged, not answered: the versions do not connect (docs/CAROUSEL-ARQ.md,
414 * "Wire formats").
415 *
416 * The 1.9.x frames, type CALL, carried in their extension the bandwidth
417 * token, a 2-bit protocol version and a carousel bit:
418 * Version 1 was 1.9.18. Version 2 moved MFSK's tones from 1531..2500 Hz to
419 * 1015.6..1984.4 Hz, centred on 1500 Hz like the codec2 modes; the frames are
420 * the same, but a pair of versions 1 and 2 would connect on DATAC16 and then
421 * never hear each other at the floor, so the version keeps them apart.
422 * Version 3 (after 1.9.19) gave a BW500 session its own floor, the 500 Hz
423 * MFSK (arq_floor_mode), and was the last value the two bits held.
424 * Mercury 2.0 took another packet type (CALL2, above).
425 *
426 * Minor 1 added the TX-MFSK-only bit, a station capability: "I can only
427 * transmit MFSK" (arq:mfsk_tx_only; an FSK-only radio). 1.9.20 sent minor
428 * 0 in 3 bits and reads any minor, so it reads a minor-1 frame as before,
429 * the bit as part of the minor; a station of minor 1 or later reads 1.9.20's
430 * frames as before too. Only a station that sets the bit needs a peer that
431 * reads it, so it ignores frames of minor 0 (arq_protocol_connect_minor).
432 *
433 * CALL 1: 0 | session id 2-3: CRC16 of the called callsign
434 * 4-13: the caller's callsign, arithmetic-coded
435 * ACCEPT 1: 1 | session id 2-3: CRC16 of the caller's callsign
436 * 4-5: CRC16 of the called callsign 6-7: the callee's nonce
437 * 8: the caller's start rung (3 bits), the callee hears the caller
438 * below the control mode (bit 3), 4 bits zero
439 * 9: the SNR the callee read the CALL at, dB + 128 (0: unknown)
440 * 10-13: zero */
441#define ARQ_CONNECT_MINOR 1
442#define ARQ_CONNECT_EXT_BW_MASK 0x03
443#define ARQ_CONNECT_EXT_MINOR_SHIFT 2
444#define ARQ_CONNECT_EXT_MINOR_MASK 0x0C
