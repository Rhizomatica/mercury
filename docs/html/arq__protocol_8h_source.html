17/* ======================================================================
18 * Plain session frame header (8 bytes): the DISCONNECT's. A session's other
19 * frames are the carousel's (carousel.h), CALL/ACCEPT have their own layout
20 * (below).
21 *
22 * Byte 0: framer byte — set/read by write_frame_header()/parse_frame_header()
23 * bits [7:5] = packet_type (PACKET_TYPE_ARQ_CONTROL)
24 * bits [4:0] = extension field (0)
25 * Byte 1: subtype — arq_subtype_t
26 * Byte 2: flags — 0
27 * Byte 3: session_id — the session's id, chosen by the caller
28 * Byte 4: tx_seq — 0
29 * Byte 5: rx_ack_seq — 0
30 * Byte 6: snr_raw — local RX SNR; 0=unknown, else round(dB) + 128
31 * Byte 7: ack_delay — 0
32 * ====================================================================== */
33
34#define ARQ_HDR_SUBTYPE_IDX 1
35#define ARQ_HDR_FLAGS_IDX 2
36#define ARQ_HDR_SESSION_IDX 3
37#define ARQ_HDR_SEQ_IDX 4
38#define ARQ_HDR_ACK_IDX 5
39#define ARQ_HDR_SNR_IDX 6
40#define ARQ_HDR_DELAY_IDX 7
41#define ARQ_FRAME_HDR_SIZE 8 /* bytes 0-7 inclusive */

43/* CONNECT frames (CALL/ACCEPT) compact layout — 14 bytes, DATAC16 only.
44 * Uses PACKET_TYPE_ARQ_CALL in the framer byte.
45 *
46 * Byte 0: framer byte (PACKET_TYPE_ARQ_CALL | BW token, set by write_frame_header)
47 * Byte 1: connect_meta = (session_id & 0x7F) | (is_accept ? 0x80 : 0x00)
48 * Bytes 2-3: CRC16-CCITT of DST callsign (little-endian) — for local validation
49 * Bytes 4-13: arithmetic_encode(SRC callsign only) — 10 bytes, fits callsigns up to ~14 chars
50 */
51#define ARQ_CONNECT_SESSION_IDX 1
52#define ARQ_CONNECT_PAYLOAD_IDX 2
53#define ARQ_CONNECT_SESSION_MASK 0x7F
54#define ARQ_CONNECT_ACCEPT_FLAG 0x80
55#define ARQ_CONTROL_FRAME_SIZE 14
56#define ARQ_CONNECT_META_SIZE 2 /* framer byte + connect_meta byte */
57#define ARQ_CONNECT_MAX_ENCODED (ARQ_CONTROL_FRAME_SIZE - ARQ_CONNECT_META_SIZE)
58#define ARQ_CONNECT_DST_CRC_SIZE 2 /* CRC16-CCITT of DST at bytes [2..3], little-endian */
59#define ARQ_CONNECT_SRC_MAX_ENCODED (ARQ_CONNECT_MAX_ENCODED - ARQ_CONNECT_DST_CRC_SIZE) /* 10 */
60
61/* Compact CQ frame layout — 14 bytes, DATAC16 only.
62 * Uses PACKET_TYPE_ARQ_CQ in the framer byte.
63 *
64 * Byte 0: framer byte (PACKET_TYPE_ARQ_CQ | BW token)
65 * Bytes 1-13: arithmetic_encode(SRC callsign only)
66 */
67#define ARQ_CQ_PAYLOAD_IDX 1
68#define ARQ_CQ_SRC_MAX_ENCODED (ARQ_CONTROL_FRAME_SIZE - ARQ_CQ_PAYLOAD_IDX) /* 13 */

118/* The one FreeDV mode used for the ARQ control frames (CALL/ACCEPT/

125/* Timing constants shared across modules. The two guards are measured, not
126 * tuned, and are no longer configuration: they bundle the radio's TX->RX
127 * switch with the audio backend's buffering, which nobody can tune blind. */
128#define ARQ_CHANNEL_GUARD_MS 700 /* IRS response guard after frame decode.
129 * OFDM decode fires ~200ms before sender
130 * PTT-OFF, so effective gap at sender is
131 * (guard - 200ms) ~= 500ms. Radio needs
132 * ~340ms for TX->RX switch -> 160ms margin
133 * for preamble detection. At 500ms the
134 * effective gap was ~300ms, causing ~50%
135 * ACK loss on DATAC1 (< 340ms switch). */
136
137#define ARQ_ISS_POST_ACK_GUARD_MS 900 /* ISS guard before resuming DATA TX
138 * after receiving an ACK from the IRS.
139 * Larger than ARQ_CHANNEL_GUARD_MS:
140 * ack_rx fires ~168ms before IRS PTT-OFF,
141 * so the effective gap at IRS is only
142 * (guard + 100ms head) - 168ms.
143 * At 500ms: gap=432ms, too tight for
144 * DATAC1 re-sync after IRS ACK TX.
145 * At 900ms: gap=832ms -- 492ms of clear
146 * air before the DATAC1 preamble. */
147#define ARQ_ACCEPT_RX_WINDOW_MS 9000 /* ACCEPTING RX window after ACCEPT TX:
148 * ISS_guard(900)+DATAC15(4400)+margin(3700)
149 * Old value 7000 left only ~300ms margin
150 * and raced with TIMER_RETRY, causing
151 * 3-4 wasted ACCEPT retries (~28s). */
152/* Grace period after entering CALLING/ACCEPTING (i.e. after PENDING is sent)
153 * during which LISTEN OFF is deferred. Scanning hosts (BPQ32) need a moment to
154 * process PENDING and cancel their own dwell timers; without this grace, a
155 * LISTEN OFF that crosses PENDING on the wire cancels the handshake — the host
156 * moves to the next channel and the reply goes out on the wrong frequency.
157 * 2 s is conservative: ~0.5 s of TCP+host processing with headroom. */
158#define ARQ_LISTEN_OFF_GRACE_MS 2000
159#define ARQ_CALL_RETRY_SLOTS_DEFAULT 4 /* CALL retries before giving up */
160/* CALLs on the fast control mode before the caller escalates to the MFSK
161 * floor; after that it alternates MFSK and the control mode (#235: under
162 * fading an MFSK CALL is no surer than a DATAC16 one at -9..-11 dB, and all-
163 * MFSK lost connects DATAC16 made; MFSK is what connects at -13 dB). A good
164 * link connects exactly as quickly as before. */
165#define ARQ_CALL_FAST_SLOTS 2
166/* After an ACCEPT on MFSK the callee listens this long for the caller's first
167 * round: the caller decodes the ACCEPT ~4 s after it ends, keys one ISS guard
168 * later, and a floor round is a 13.5 s frame the callee decodes ~4 s after. */
169#define ARQ_ACCEPT_RX_WINDOW_FLOOR_MS 26000
170#define ARQ_ACCEPT_RETRY_SLOTS_DEFAULT 4 /* ACCEPT retries before returning */
171#define ARQ_DISCONNECT_RETRY_SLOTS_DEFAULT 2 /* DISCONNECT frame retries */

207/* The silence between two frames of one keydown (the carousel's poll and its
208 * first burst). The peer is already receiving, so this is not a turnaround:
209 * it only has to let the peer's control decoder report the first frame and
210 * its payload decoder drop any sync taken off it before the next preamble
211 * arrives. Measured, not tuned: see the one-keydown bench. */
212#define ARQ_ACK_DATA_GAP_MS 300
213
214/* Listen before talk. "Busy" is a decoder holding sync on an incoming burst
215 * (arq_update_link_metrics()); HOLD_MS is how stale a sync observation may be
216 * and still count as busy. The modem updates it once per RX chunk, so this
217 * only covers scheduling jitter, not a whole frame: sync keeps being
218 * re-observed for as long as the burst lasts.
219 *
220 * A deferred transmission (the DISCONNECT) waits in DEFER_MS steps, at most
221 * DEFER_MAX of them: false sync is real (a payload decoder can latch onto a
222 * control burst), and a decoder stuck in sync must not block a transmission
223 * forever, so after that we key anyway and take the collision. */
224#define ARQ_CHANNEL_SYNC_HOLD_MS 250
225#define ARQ_CHANNEL_DEFER_MS 500
226#define ARQ_CHANNEL_DEFER_MAX 20 /* ~10 s, then key regardless */
227/* Before a DISCONNECT: how long past the longer reply guard to leave for a
228 * frame our last transmission invited to start and be heard, before keying
229 * over the gap. */
230#define ARQ_REPLY_WINDOW_MARGIN_MS 1000

351/* --- CALL/ACCEPT, protocol version 1 (PACKET_TYPE_ARQ_CALL) ---
352 *
353 * 14 bytes (a DATAC16 frame), padded to the frame on MFSK. Byte 0 is the
354 * framer: type CALL, and in its extension the bandwidth token (2 bits), the
355 * protocol version (2 bits) and the carousel bit. A station takes a frame
356 * only with this version and the carousel bit; a 1.9.x station reads that
357 * extension as an unknown bandwidth token and drops it, so the versions do
358 * not connect (docs/CAROUSEL-ARQ.md, "Wire formats").
359 *
360 * CALL 1: 0 | session id 2-3: CRC16 of the called callsign
361 * 4-13: the caller's callsign, arithmetic-coded
362 * ACCEPT 1: 1 | session id 2-3: CRC16 of the caller's callsign
363 * 4-5: CRC16 of the called callsign 6-7: the callee's nonce
364 * 8: the caller's start rung (3 bits; 7 = ARQ_ACCEPT_LEVEL_CTL_DEAF),
365 * 5 bits zero 9-13: zero */
366#define ARQ_CONNECT_VERSION 1
367#define ARQ_CONNECT_EXT_BW_MASK 0x03
368#define ARQ_CONNECT_EXT_VERSION_SHIFT 2
369#define ARQ_CONNECT_EXT_VERSION_MASK 0x0C
370#define ARQ_CONNECT_EXT_CAROUSEL 0x10
371#define ARQ_ACCEPT_CALLED_CRC_IDX 4
372#define ARQ_ACCEPT_NONCE_IDX 6
373#define ARQ_ACCEPT_LEVEL_IDX 8
374#define ARQ_ACCEPT_LEVEL_CTL_DEAF 7 /* the callee hears the caller below
375 the control mode: start at the floor */
