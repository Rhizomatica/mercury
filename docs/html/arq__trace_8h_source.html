1/* Low-perturbation event trace for diagnosing connect-path races.
2 *
3 * The failure this exists for disappears under -v: any per-event logging shifts
4 * the timing enough to hide it. So nothing here does I/O on the hot path. A
5 * record is 16 bytes appended to a preallocated array under one relaxed atomic
6 * increment; the whole trace is printed once, later, by arq_trace_dump().
7 *
8 * SPDX-License-Identifier: GPL-3.0-or-later
9 */
10#ifndef ARQ_TRACE_H_
11#define ARQ_TRACE_H_

13#include <stdint.h>
14#include <stdbool.h>

16typedef enum {
17 ARQ_TR_RX_FRAME = 1, /* a frame decoded off the air: a=packet_type b=subtype */
18 ARQ_TR_EV_IN, /* event entering the FSM: a=event id b=conn_state */
19 ARQ_TR_EV_OUT, /* FSM returned: a=conn_state */
20 ARQ_TR_TX_START, /* we keyed up: a=packet_type b=mode&0xff */
21 ARQ_TR_TX_END, /* we unkeyed */
22 ARQ_TR_DROP, /* a frame was discarded: a=reason */
23 ARQ_TR_DECODE, /* decoder said something: a=mode b=sync|status<<1 x=bytes */
24} arq_trace_kind_t;
